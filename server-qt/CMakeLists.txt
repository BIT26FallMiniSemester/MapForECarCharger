# 配置 Qt6 服务端、SQLite、Socket 资源、测试目标及安装规则。

cmake_minimum_required(VERSION 3.16)
project(charger_server VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Network Sql Test)
add_library(charger_core src/common.cpp src/database.cpp src/business.cpp src/maps.cpp src/server.cpp src/forecastworker.cpp src/dashboardserver.cpp src/analyticsresult.cpp)
target_include_directories(charger_core PUBLIC src)
target_link_libraries(charger_core PUBLIC Qt6::Core Qt6::Network Qt6::Sql)
target_compile_options(charger_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
qt_add_resources(charger_core resources PREFIX / FILES migrations/001_initial.sql migrations/002_catalog_details.sql)
qt_add_resources(charger_core socket_contract PREFIX / BASE ../contracts FILES ../contracts/schemas/socket.json)
qt_add_resources(charger_core dashboard PREFIX / BASE . FILES web/dashboard.html)
add_executable(charger-server src/main.cpp)
target_link_libraries(charger-server PRIVATE charger_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(charger-tests tests/test_server.cpp)
  target_link_libraries(charger-tests PRIVATE charger_core Qt6::Test)
  target_compile_options(charger-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME charger-tests COMMAND charger-tests)
  add_executable(analytics-tests tests/test_analytics.cpp)
  target_link_libraries(analytics-tests PRIVATE charger_core Qt6::Test)
  add_test(NAME analytics-tests COMMAND analytics-tests)
endif()
install(TARGETS charger-server RUNTIME DESTINATION bin)
